    <script>
        const products = [
            { id: 1, platform: "IOS", name: "🔮 FLUORITE IOS", price: 20 },
            { id: 2, platform: "IOS", name: "🎱 FLUORITE 8BALL", price: 20 },
            { id: 3, platform: "IOS", name: "🌟 MIGUL PRO", price: 25 },
            { id: 4, platform: "IOS", name: "🌟 MIGUL BASIC", price: 15 },
            { id: 5, platform: "IOS", name: "📡 PROXY ALL PORT", price: 10 },
            { id: 6, platform: "IOS", name: "✒️ CERTIFICATE", price: 5 },
            { id: 7, platform: "IOS", name: "💠FILES VIP FILZA", price: 30 },
            { id: 8, platform: "ANDROID", name: "✅DRIP CLIENT — ROOT / NO ROOT", price: 25 },
            { id: 9, platform: "ANDROID", name: "✅PROXY DRIP CLIENT", price: 15 },
            { id: 10, platform: "ANDROID", name: "👑HG CHEAT", price: 40 },
            { id: 11, platform: "ANDROID", name: "🌟BR MODS", price: 20 },
            { id: 12, platform: "ANDROID", name: "🛒PATO TEAM — BLUE / ORANGE", price: 35 },
            { id: 13, platform: "ANDROID", name: "🐶TM PANEL", price: 30 }
        ];

        let db = JSON.parse(localStorage.getItem('storeDB')) || {
            users: { "admin": { pass: "admin", balance: 1000 } }
        };

        let currentUser = "";

        function saveDB() {
            localStorage.setItem('storeDB', JSON.stringify(db));
        }

        function renderProducts() {
            const iosList = document.getElementById('iosList');
            const androidList = document.getElementById('androidList');
            iosList.innerHTML = ''; androidList.innerHTML = '';

            products.forEach(p => {
                const html = `
                    <div class="product">
                        <div class="p-name">${p.name}</div>
                        <div style="display:flex; align-items:center; gap:10px;">
                            <div style="color:var(--green); font-weight:bold;">$${p.price}</div>
                            <button class="buy-btn" onclick="buy(${p.id})">شراء</button>
                        </div>
                    </div>
                `;
                if (p.platform === 'IOS') iosList.innerHTML += html;
                else androidList.innerHTML += html;
            });
        }
        renderProducts();

        function login() {
            const user = document.getElementById('u').value;
            const pass = document.getElementById('p').value;
            
            if (db.users[user] && db.users[user].pass === pass) {
                currentUser = user;
                document.getElementById('loginBox').style.display = 'none';
                document.getElementById('storeBox').style.display = 'block';
                document.getElementById('cUser').innerText = user;
                updateBalanceUI();
            } else {
                alert("بيانات خاطئة!");
            }
        }

        function updateBalanceUI() {
            document.getElementById('cBalance').innerText = db.users[currentUser].balance;
        }

        function buy(id) {
            const product = products.find(p => p.id === id);
            if (db.users[currentUser].balance >= product.price) {
                db.users[currentUser].balance -= product.price;
                saveDB();
                updateBalanceUI();
                alert(`✅ تم شراء ${product.name} بنجاح!\nتواصل مع @ASSTAXIT☠️ للاستلام.`);
            } else {
                alert("❌ رصيدك غير كافٍ!");
            }
        }

        function toggleAdmin() {
            if (currentUser !== 'admin') return alert("إنت مش أدمن!");
            const admin = document.getElementById('adminPanel');
            admin.style.display = admin.style.display === 'block' ? 'none' : 'block';
        }

        function addBalance() {
            const target = document.getElementById('targetUser').value;
            const amount = parseFloat(document.getElementById('addAmount').value);

            if (currentUser !== 'admin') return alert("إنت مش أدمن!");
            if (!db.users[target]) return alert("المستخدم غير موجود!");
            if (isNaN(amount)) return alert("اكتب مبلغ صحيح!");

            db.users[target].balance += amount;
            saveDB();
            alert(`تم شحن $${amount} لـ ${target}`);
            if (target === currentUser) updateBalanceUI();
        }
